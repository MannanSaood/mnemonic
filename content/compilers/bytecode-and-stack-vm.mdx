        type Instr struct {
            Op  Op  // OpConst, OpLoad, OpGt, OpDiv, ..., OpJumpIfFalseOrPop, OpReturn
            Arg int // constant index, slot index or jump target
        }

        type Chunk struct {
            Code   []Instr
            Consts []Value
            Slots  []string // metric names, turned into indexes at compile time
        }

        func (c *Chunk) compile(e Expr) {
            switch e := e.(type) {
            case *Leaf:
                if e.Tok.Kind == Name {
                    c.emit(OpLoad, c.slot(e.Tok.Text))
                } else {
                    c.emit(OpConst, c.constant(e.Value))
                }
            case *Binary:
                c.compile(e.Left)
                if jumpOp, ok := shortCircuit[e.Op]; ok { // "and", "or"
                    j := c.emit(jumpOp, -1)
                    c.compile(e.Right)
                    c.Code[j].Arg = len(c.Code) // patch: jump to just past the right side
                    return
                }
                c.compile(e.Right)
                c.emit(binaryOps[e.Op], 0)
            }
        }

        func run(c *Chunk, slots []Value) (Value, error) {
            stack := make([]Value, 0, 16)
            for ip := 0; ; {
                in := c.Code[ip]
                ip++
                switch in.Op {
                case OpConst:
                    stack = append(stack, c.Consts[in.Arg])
                case OpLoad:
                    stack = append(stack, slots[in.Arg])
                case OpJumpIfFalseOrPop, OpJumpIfTrueOrPop:
                    if stack[len(stack)-1] == (in.Op == OpJumpIfTrueOrPop) {
                        ip = in.Arg // already decided: keep the value, skip the right side
                    } else {
                        stack = stack[:len(stack)-1]
                    }
                case OpReturn:
                    return stack[len(stack)-1], nil
                default: // binary: pop two, push one
                    a, b := stack[len(stack)-2], stack[len(stack)-1]
                    v, err := apply(in.Op, a, b)
                    if err != nil {
                        return nil, err
                    }
                    stack = append(stack[:len(stack)-2], v)
                }
            }
        }

        # vm.py: compile each rule once to bytecode, then run it in a loop
        from dataclasses import dataclass, field
        from enum import Enum, auto

        class Op(Enum):
            CONST = auto()                 # push consts[arg]
            LOAD = auto()                  # push slots[arg]
            BINARY = auto()                # pop b, pop a, push arg(a, b)
            JUMP_IF_FALSE_OR_POP = auto()  # top false: jump to arg, keep it; else pop it
            JUMP_IF_TRUE_OR_POP = auto()
            RETURN = auto()

        @dataclass
        class Chunk:
            code: list[tuple[Op, object]] = field(default_factory=list)
            consts: list[Value] = field(default_factory=list)
            slots: list[str] = field(default_factory=list)  # metric names, indexed at compile time

            def compile(self, e: Expr) -> None:
                match e:
                    case Leaf(tok) if tok.kind == "NAME":
                        self.code.append((Op.LOAD, self.slot(tok.text)))
                    case Leaf():
                        self.code.append((Op.CONST, self.constant(e.value)))
                    case Binary(op="and" | "or" as op, left=l, right=r):
                        self.compile(l)
                        jump = len(self.code)
                        self.code.append((Op.JUMP_IF_FALSE_OR_POP if op == "and" else Op.JUMP_IF_TRUE_OR_POP, None))
                        self.compile(r)
                        self.code[jump] = (self.code[jump][0], len(self.code))  # patch: just past the right side
                    case Binary(op=op, left=l, right=r):
                        self.compile(l)
                        self.compile(r)
                        self.code.append((Op.BINARY, OPERATORS[op]))  # operator.gt, operator.truediv, ...

        def run(chunk: Chunk, slots: list[Value]) -> Value:
            stack: list[Value] = []
            ip = 0
            while True:
                op, arg = chunk.code[ip]
                ip += 1

                    case Op.CONST:
                        stack.append(chunk.consts[arg])

                        stack.append(slots[arg])

        type Jump = { op: "jumpIfFalseOrPop" | "jumpIfTrueOrPop"; target: number };
        type Instr =
          | { op: "const"; arg: number } // push consts[arg]
          | { op: "load"; arg: number } // push slots[arg]
          | { op: "binary"; fn: (a: Value, b: Value) => Value }
          | Jump
          | { op: "return" };

        interface Chunk { code: Instr[]; consts: Value[]; slots: string[] }

        function compile(e: Expr, c: Chunk): void {
          if (e.kind === "leaf") {
            if (e.token.kind === "name") c.code.push({ op: "load", arg: slotOf(c, e.token.text) });
            else c.code.push({ op: "const", arg: c.consts.push(e.value) - 1 });
            return;
          }
          compile(e.left, c);
          if (e.op === "and" || e.op === "or") {
            const jump: Jump = { op: e.op === "and" ? "jumpIfFalseOrPop" : "jumpIfTrueOrPop", target: -1 };
            c.code.push(jump);
            compile(e.right, c);
            jump.target = c.code.length; // patch: just past the right side
            return;
          }
          compile(e.right, c);
          c.code.push({ op: "binary", fn: OPERATORS[e.op] });
        }

        export function run(c: Chunk, slots: Value[]): Value {
          const stack: Value[] = [];
          for (let ip = 0; ; ) {
            const ins = c.code[ip++];
            switch (ins.op) {
              case "const": stack.push(c.consts[ins.arg]); break;
              case "load": stack.push(slots[ins.arg]); break;
              case "binary": { const b = stack.pop()!; stack.push(ins.fn(stack.pop()!, b)); break; }

                else stack.pop();
                break;
              case "return": return stack.pop()!;
            }
          }
        }

        // vm.rs: compile each rule once to bytecode, then run it in a loop
        #[derive(Clone, Copy)]
        pub enum Instr {
            Const(u16),            // push consts[i]
            Load(u16),             // push slots[i]
            Binary(BinOp),         // pop b, pop a, push a op b
            JumpIfFalseOrPop(u32), // top false: jump, keeping it; else pop it
            JumpIfTrueOrPop(u32),
            Return,
        }

        impl Chunk {
            fn compile(&mut self, e: &Expr) {
                match e {
                    Expr::Name(n) => { let s = self.slot(n); self.code.push(Instr::Load(s)) }
                    Expr::Lit(v) => { let k = self.constant(v.clone()); self.code.push(Instr::Const(k)) }
                    Expr::Binary(op @ ("and" | "or"), l, r) => {
                        self.compile(l);
                        let j = self.code.len();
                        self.code.push(Instr::Return); // placeholder, patched below
                        self.compile(r);
                        let past = self.code.len() as u32;
                        self.code[j] = if *op == "and" { Instr::JumpIfFalseOrPop(past) } else { Instr::JumpIfTrueOrPop(past) };
                    }
                    Expr::Binary(op, l, r) => {
                        self.compile(l);
                        self.compile(r);
                        self.code.push(Instr::Binary(BinOp::from(*op)));
                    }
                }
            }
        }

        pub fn run(c: &Chunk, slots: &[Value]) -> Result<Value, EvalError> {
            let mut stack: Vec<Value> = Vec::with_capacity(16);
            let mut ip = 0;
            loop {
                let ins = c.code[ip];
                ip += 1;
                match ins {
                    Instr::Const(i) => stack.push(c.consts[i as usize].clone()),
                    Instr::Load(i) => stack.push(slots[i as usize].clone()),
                    Instr::Binary(op) => {
                        let b = stack.pop().unwrap();
                        let a = stack.pop().unwrap();
                        stack.push(op.apply(a, b)?);

                    Instr::JumpIfFalseOrPop(t) | Instr::JumpIfTrueOrPop(t) => {
                        let jump_on = matches!(ins, Instr::JumpIfTrueOrPop(_));

                            ip = t as usize; // already decided: keep the value, skip the right side
                        } else {
                            stack.pop();
                        }
                    }
                    Instr::Return => return Ok(stack.pop().unwrap()),
                }
            }
        }

      - text: It pops two values, compares them, and pushes true or false.
        correct: true
        why: Right. Operands come from the top of the stack and the result goes back on it, so the instruction needs no operand of its own.
      - text: It jumps to another instruction if the comparison is true.
        why: Jumps are separate instructions. GT only computes a boolean, and a later jump instruction may look at it.
      - text: It walks the left and right children of a comparison node.
        why: That is the tree walker. Bytecode has no children to walk; the operands were pushed by earlier instructions.
  engineer:
    question: How must and compile so that requests > 0 and errors / requests > 0.05 never divides by zero?
    options:
      - text: As a conditional jump after the left side that skips the right side when the left is false, with its target patched once the right side is compiled.
        correct: true
        why: Right. The jump is the bytecode form of short-circuit evaluation. Its target is only known after the right side is emitted, hence the patch.
      - text: As an AND instruction after both sides, like the other binary operators.
        why: Then both sides run before AND looks at them, and the division runs on an idle service. That is the break-it mode in the playground.
      - text: As a check inside DIV that returns 0 when the divisor is 0.
        why: That hides a real error everywhere else, and it still evaluates code the rule says not to evaluate.
  deep:
    question: Python 3.11 made its bytecode VM substantially faster without becoming a register machine or adding a JIT. What was the main technique?
    options:
      - text: A specializing adaptive interpreter. Instructions are rewritten in place into versions specialized for the types actually seen, such as adding two ints.
        correct: true
        why: Right. That is PEP 659. Generic instructions such as BINARY_OP check types on every run; the specialized forms skip most of that while the types stay the same.
      - text: It switched the VM from a stack to registers.
        why: CPython is still a stack VM. Register VMs, such as Lua's, run fewer instructions, but that change was not part of 3.11.
      - text: It compiles bytecode to machine code the first time a function runs.
        why: That is a JIT. CPython 3.13 added an experimental one, disabled by default, after the 3.11 speedups.

    - role: Chunk

    - role: Instruction pointer

      job: Index of the next instruction. Jumps set it.
    - role: Value stack

      job: Holds operands and intermediate results. Binary instructions pop two and push one.
    - role: Dispatch loop

      job: Fetch, advance ip, switch on the opcode, repeat until RETURN.
  useWhen:
    - The same program runs many times on different inputs, such as rules per series, queries per row or scripts per frame.

  avoidWhen:

## [concept] A flat list and a loop

**Compile the syntax tree once into a flat list of instructions, then run it in one loop with an <Term id="instruction-pointer">instruction pointer</Term> and a value stack.**

`requests > 0 and errors / requests > 0.05` becomes ten instructions: push `requests`, push `0`, compare, jump if false, push `errors`, push `requests`, divide, push `0.05`, compare, return. The VM runs them in order, except where a jump says otherwise.

<Depth level={2}>
The list is <Term id="bytecode">bytecode</Term>, and the VM is a <Term id="stack-machine">stack machine</Term>. Each instruction takes its operands from the top of a stack and pushes its result, so `GT` needs no arguments. The compiler produces this order by walking the <Term id="abstract-syntax-tree">tree</Term> in post-order: left child, right child, then the operator. A chunk holds three things: the instructions, a table of constants they refer to by index, and the slots for metric names.
</Depth>

<Depth level={3}>
Crafting Interpreters builds the same language twice, as a tree walker in Java and as a bytecode VM in C, and the VM is much faster. Part of that is C. The rest is the design: a contiguous array instead of objects linked by pointers, a switch instead of a virtual call per node, and names resolved before the program runs.
</Depth>

## [problem] A tree walk per series

The alerting service evaluates each rule by walking its tree: a recursive function switches on the node type, recurses into the children, and looks each metric up by name in the series' map.

At launch, rules ran per service, a few hundred evaluations every 10 seconds. The walker was clear and correct, and nobody profiled it.

<Depth level={2}>
Every evaluation repeats work that never changes between series: the same type switches, the same pointer chases to the same children, the same map lookups of the same names. Only the metric values differ.
</Depth>

<Depth level={3}>
Each node is a separate heap object, so a walk jumps around memory. Each visit is a call with a stack frame. Each name is hashed and compared on every lookup. None of this is slow once. All of it is paid on every evaluation.
</Depth>

## [pain] Every rule, every series

**Change request:** every alert rule must now run against every series, every 10 seconds, 200 rules across 50,000 series.

That is 10 million evaluations per cycle, and each one walks the whole tree again. The marked lines are the loop that now runs 10 million times and the walk inside it. The evaluator's CPU climbs, the cycle overruns its 10 seconds, and alerts fire late.

<Depth level={2}>
Adding machines spreads the work but does not reduce it. Caching results does not help either, because every series has different values. The repeated part is the interpretation of the rule's structure, and that can be done once.
</Depth>

<Depth level={3}>
Profile before rewriting. If the time goes to fetching metrics or to allocation, bytecode will not fix it. If it goes to the walk itself (calls, type dispatch, hashing names), compiling the rule once is the standard fix.
</Depth>

## [morph] Compile once, run a loop

Drag the slider. The tree walker splits in two: a compiler that runs once when a rule is saved, and a VM loop that runs for every series. The rule's structure is interpreted once, into a chunk of instructions.

Hover a box in the diagram to see its code.

<Depth level={2}>
Names turn into slot numbers at compile time. The engine fills a slot array from each series once, and the VM reads it by index. `and` and `or` do not become instructions that combine two values. They become jumps, because the right side must not run when the left side has already decided the answer.
</Depth>

<Depth level={3}>
<Lang only="go">An instruction is an opcode and an int, so the code is one contiguous slice. The `switch` in `run` is the dispatch loop. Go compiles a dense switch on a small integer type into a jump table or a binary search, so dispatch costs a few machine instructions.</Lang>
<Lang only="python">Each instruction is a tuple, and binary operators carry the function from the `operator` module, so the loop calls `operator.truediv` directly. Keep in mind that this loop is itself run by CPython's VM. Measure it against the tree walker before you ship it.</Lang>
<Lang only="typescript">The instruction union is discriminated by `op`, so the `switch` in `run` narrows each case and the compiler checks the operands. Binary instructions hold the operator's function itself, a small step toward closure compilation.</Lang>
<Lang only="rust">`Instr` is a small `Copy` enum, so `c.code[ip]` copies it out without borrowing the chunk, and the `match` usually compiles to a jump table. The placeholder pushed before the jump is overwritten once the target is known, which is <Term id="jump-patching">jump patching</Term>.</Lang>
</Depth>

## [walk] Compile and run the guarded ratio

`requests > 0 and errors / requests > 0.05`, compiled once and run on an idle service. Step through it with the numbered buttons.

<Depth level={3}>
The jump keeps the value it tested when it jumps and pops it when it falls through. So after `and` the stack holds exactly one boolean either way, which is the stack effect a binary operator would have. The compiler can track the stack depth of every instruction this way and size the stack before running. CPython stores that maximum per code object as `co_stacksize`.
</Depth>

## [again] Every rule, every series, compiled

The same load, with the rules compiled. Saving a rule runs the compiler once. Each of the 10 million evaluations per cycle runs a short loop over an array, reads metrics by slot, and skips the right side of `and` and `or` with a jump. There is no recursion and no lookup by name.

<Depth level={2}>
Keep the tree walker. It is the reference implementation: run both on recorded series in tests and compare the answers, exactly as the playground does. Keep a source span on each instruction too, so a runtime error such as division by zero points at the right part of the rule.
</Depth>

<Depth level={3}>
Beyond this point the gains come from fewer instructions and cheaper dispatch. A register VM, as in Lua since version 5.0, does more work per instruction. Superinstructions fuse common pairs such as `LOAD` then `CONST` then `GT`. Specialization, as in CPython 3.11, rewrites generic instructions for the types it actually sees.
</Depth>

### The instruction set used here

| Opcode | Operand | Stack effect |
|---|---|---|
| `CONST` | constant index | push 1 |
| `LOAD` | slot index | push 1 |
| `GT`, `DIV`, `EQ`, … | none | pop 2, push 1 |
| `NOT`, `NEG` | none | pop 1, push 1 |
| `JUMP_IF_FALSE_OR_POP` | target | keep 1 and jump, or pop 1 |
| `JUMP_IF_TRUE_OR_POP` | target | keep 1 and jump, or pop 1 |
| `RETURN` | none | pop 1 and stop |

The two conditional jumps are modelled on instructions CPython used for `and` and `or` before version 3.12.
