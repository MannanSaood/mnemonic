        // lexer.go: read the rule once, left to right, and hand out tokens
        type Token struct {
            Kind       Kind   // Name, Keyword, Number, String, Op, Error, EOF
            Text       string
            Start, End int    // byte offsets, for error positions
        }

        type Lexer struct {
            src string
            pos int // next byte to read
        }

        func (l *Lexer) peek() byte {
            if l.pos >= len(l.src) {
                return 0
            }
            return l.src[l.pos]
        }

        func (l *Lexer) Next() Token {
            for l.peek() == ' ' || l.peek() == '\t' || l.peek() == '\n' {
                l.pos++
            }
            start := l.pos
            tok := func(k Kind) Token { return Token{k, l.src[start:l.pos], start, l.pos} }
            if l.pos >= len(l.src) {
                return tok(EOF)
            }
            c := l.src[l.pos]
            l.pos++
            switch {
            case strings.IndexByte("<>=!", c) >= 0:
                if l.peek() == '=' { // longest match: ">=" is one token, not two
                    l.pos++
                    return tok(Op)
                }
                if c == '=' || c == '!' {
                    return tok(Error) // a lone = or !
                }
                return tok(Op)
            case strings.IndexByte("+-*/()", c) >= 0:
                return tok(Op)
            case c == '"':
                for l.peek() != '"' && l.peek() != '\n' && l.peek() != 0 {
                    l.pos++
                }
                if l.peek() != '"' {
                    return tok(Error) // unterminated: stop at the end of the line
                }
                l.pos++
                return tok(String)
            case isDigit(c):
                for isDigit(l.peek()) || l.peek() == '.' {
                    l.pos++
                }
                return tok(Number)
            case isLetter(c):
                for isLetter(l.peek()) || isDigit(l.peek()) {
                    l.pos++
                }
                if keywords[l.src[start:l.pos]] {
                    return tok(Keyword)
                }
                return tok(Name)
            }
            return tok(Error) // no token starts with this character
        }

        # lexer.py: one pass over the rule, yielding tokens with positions
        import re
        from dataclasses import dataclass
        from typing import Iterator

        KEYWORDS = {"and", "or", "not", "true", "false"}
        TOKEN_RE = re.compile(r"""
              (?P<SKIP>\s+|\#[^\n]*)
            | (?P<NUMBER>\d+(?:\.\d+)?)
            | (?P<NAME>[A-Za-z_]\w*)
            | (?P<STRING>"[^"\n]*")
            | (?P<UNTERMINATED>"[^"\n]*)
            | (?P<OP>[<>=!]=|[<>+\-*/()])
            | (?P<BAD>.)
        """, re.VERBOSE)

        @dataclass(frozen=True)
        class Token:
            kind: str
            text: str
            start: int
            end: int

        def tokens(rule: str) -> Iterator[Token]:
            for m in TOKEN_RE.finditer(rule):
                kind, text = m.lastgroup, m.group()
                if kind == "SKIP":
                    continue
                if kind == "NAME" and text in KEYWORDS:
                    kind = "KEYWORD"
                if kind in ("UNTERMINATED", "BAD"):
                    raise RuleError(f"{kind.lower()} {text!r} at offset {m.start()}")
                yield Token(kind, text, m.start(), m.end())
            yield Token("EOF", "", len(rule), len(rule))

        // lexer.ts: one pass over the rule, yielding tokens with positions
        type Kind = "name" | "keyword" | "number" | "string" | "op" | "error" | "eof";
        interface Token { kind: Kind; text: string; start: number; end: number }

        const KEYWORDS = new Set(["and", "or", "not", "true", "false"]);
        const isDigit = (c: string) => c >= "0" && c <= "9";
        const isWord = (c: string) => /^[A-Za-z0-9_]$/.test(c);

        export function* lex(src: string): Generator<Token> {
          let pos = 0;
          const peek = () => src[pos] ?? "";
          for (;;) {
            while (/^\s$/.test(peek())) pos++;
            const start = pos;
            const tok = (kind: Kind): Token => ({ kind, text: src.slice(start, pos), start, end: pos });
            if (pos >= src.length) { yield tok("eof"); return; }
            const c = src[pos++];
            if ("<>=!".includes(c)) {
              if (peek() === "=") pos++; // longest match: ">=" is one token, not two
              else if (c === "=" || c === "!") { yield tok("error"); continue; } // a lone = or !
              yield tok("op");
            } else if ("+-*/()".includes(c)) {
              yield tok("op");
            } else if (c === '"') {
              while (pos < src.length && peek() !== '"' && peek() !== "\n") pos++;
              if (peek() !== '"') { yield tok("error"); continue; } // unterminated: stop at end of line
              pos++;
              yield tok("string");
            } else if (isDigit(c)) {
              while (isDigit(peek()) || peek() === ".") pos++;
              yield tok("number");
            } else if (/^[A-Za-z_]$/.test(c)) {
              while (isWord(peek())) pos++;
              yield tok(KEYWORDS.has(src.slice(start, pos)) ? "keyword" : "name");
            } else {
              yield tok("error"); // no token starts with this character
            }
          }
        }

            let words: Vec<&str> = rule.split_whitespace().collect();

        use std::{iter::Peekable, str::CharIndices};

        const KEYWORDS: [&str; 5] = ["and", "or", "not", "true", "false"];

        #[derive(Debug, Clone, Copy, PartialEq)]
        pub enum Kind { Name, Keyword, Number, Str, Op, Error(&'static str) }

        pub struct Token<'a> { pub kind: Kind, pub text: &'a str, pub span: (usize, usize) }

        pub struct Lexer<'a> { src: &'a str, chars: Peekable<CharIndices<'a>> }

        impl<'a> Iterator for Lexer<'a> {
            type Item = Token<'a>;

            fn next(&mut self) -> Option<Token<'a>> {
                while self.chars.next_if(|&(_, c)| c.is_whitespace()).is_some() {}
                let (start, c) = self.chars.next()?; // None: end of input

                        if self.chars.next_if(|&(_, n)| n == '=').is_some() {

                            Kind::Error("a lone = or !")
                        } else {
                            Kind::Op

                    '+' | '-' | '*' | '/' | '(' | ')' => Kind::Op,

                            Some(_) => Kind::Str,
                            None => Kind::Error("unterminated string"),

                        while self.chars.next_if(|&(_, n)| n.is_ascii_digit() || n == '.').is_some() {}

# ---------------------------------------------------------------- how it works

  - text: The parser asks for the next token. The lexer skips spaces and comments, then remembers where this token starts.
    nodes: [client, context]
    edges: [client>context]
    role: skip
  - text: "It reads `>`. That could be the whole operator or the start of `>=`. One peek at the next character decides, and the lexer always takes the longest match."
    nodes: [context, op, strategy]
    edges: [op>strategy]
    role: munch
  - text: A quote starts a string, which runs to the closing quote, spaces and all. If the line ends first, the lexer emits an error token with its position and carries on from the next line.
    nodes: [str, err, strategy]
    edges: [str>strategy, err>strategy]
    role: string
  - text: "A letter starts a word. The lexer reads letters and digits, then looks the word up: `and` is a keyword, `cpu` is a name."
    nodes: [name, strategy]
    edges: [name>strategy]
    role: word
  - text: Each token leaves with its kind, its text and its span. The parser never sees a character again, only tokens.
    nodes: [context, strategy]
    edges: [context>strategy]
    role: emit

        why: Then every later stage would have to rediscover where names and numbers end. Grouping them once is the lexer's whole job.
  engineer:

      - text: To skip any spaces that follow the operator.

        correct: true
        why: Right. Errors become tokens with source spans. Stopping the string at the newline keeps one missing quote from swallowing the rest of the file.
      - text: Stop at the first error and report that the rule is invalid.
        why: The author then fixes one error per run, and without a position they have to hunt for it.
      - text: Treat everything up to the end of the file as the string.
        why: Then every token after line 3 vanishes into the string, and the parser reports confusing errors far from the real one.

## [concept] A lexer cuts text into tokens

**A lexer reads source text once, left to right, and groups its characters into <Term id="token">tokens</Term>, each with a kind, its text and its position.**

`cpu >= 0.9 and env == "prod"` becomes seven tokens: a name, an operator, a number, a keyword, a name, an operator and a string. The parser that comes next works with those seven pieces and never looks at a character again.

<Depth level={2}>
A lexer keeps two marks in the text: where the current token started and the next character to read. It reads one character, and that character decides the kind of token: a digit starts a number, a letter a word, a quote a string. Then it keeps reading while the characters still belong. Sometimes one character is not enough. After `>` it takes one <Term id="lookahead">lookahead</Term> peek to see whether `=` follows. Words are read the same way whether they are names or keywords. The lexer looks the word up in a keyword table afterwards.
</Depth>

<Depth level={3}>
Every token in a typical language is described by a regular expression, so a lexer is a finite automaton: it needs no stack and runs in time linear in the input. Most production compilers still write theirs by hand, because hand-written code gives better error messages and handles the awkward cases. Those cases are the tokens that depend on context, such as nested comments or `/` in JavaScript.
</Depth>

## [problem] Rules split on spaces

An alerting service lets teams write rules such as `cpu > 0.9 and env == "prod"`. The rule parser does what most first versions do. It splits the rule on whitespace and reads the words in groups of four: metric, operator, value, then `and`.

It passes every test, because every test rule was written by the engineer who wrote the parser, with a space around every operator.

<Depth level={2}>
Splitting on spaces assumes that spaces separate tokens and never appear inside one. Neither is a rule of the language. It is a habit of the people who wrote the tests.
</Depth>

<Depth level={3}>
The same assumption also hides positions. After `split()` the words have lost their offsets, so the best error message available is "word 3 is wrong".
</Depth>

## [pain] Then on-call writes the rules

**Change request:** on-call engineers can now write alert rules in the UI, and they write `cpu>=0.9 and env=="prod eu"`.

Splitting gives four words: `cpu>=0.9`, `and`, `env=="prod` and `eu"`. The parser builds a nonsense condition from the first three and rejects the rule with "expected and, got `eu"`". The person reading that message has no way to know that the real problem is a missing space.

<Depth level={2}>
Patching it fails one case at a time. Insert spaces around operators with a regular expression, and you also change the inside of strings. Then `>` gets padded inside `>=`. Then someone writes a comment. Each fix is a new special case, and none of them knows where a token begins.
</Depth>

<Depth level={3}>
The worst failures are the ones that are not rejected. `env == "prod eu"` alone splits into `env`, `==`, `"prod` and `eu"`. A looser parser that took the first three words would silently compare `env` with `"prod`, a value that never matches, and the alert would never fire.
</Depth>

## [morph] From words to tokens

Drag the slider. The splitter disappears. In its place a lexer hands out tokens one at a time, and each token says what it is and where it came from.

Hover a box in the diagram to see its code.

<Depth level={2}>
The lexer decides where a token ends by looking at characters, not at spaces. A digit cannot continue a name, so `cpu>=` ends the name at `>`. A quote runs to the next quote, so `"prod eu"` is one token. Whitespace and comments become things to skip, not separators.
</Depth>

<Depth level={3}>
<Lang only="go">The lexer holds the source and one offset. `Next` slices `l.src[start:l.pos]` for the token text, so no bytes are copied. The real Go lexer, `go/scanner`, returns positions as compact `token.Pos` integers that a `token.FileSet` turns back into file, line and column.</Lang>
<Lang only="python">One compiled regular expression is the whole lexer. Each alternative is a named group, `finditer` tries them in order at each position, and `m.lastgroup` names the one that matched. Order is what makes `>=` win over `>`. `\w` also matches non-ASCII letters, so names such as `größe` work.</Lang>
<Lang only="typescript">The lexer is a generator, so it is lazy. Each `yield` hands one token to the parser and pauses until the parser asks for the next. The token text is a `slice` of the source, and the positions are UTF-16 code unit offsets, the same units the editor uses.</Lang>
<Lang only="rust">`Peekable<CharIndices>` gives each character with its byte offset, and `next_if` consumes a character only if it matches, which is lookahead and advance in one call. Tokens borrow `&'a str` slices of the source, so the lifetime ties every token to the text it came from.</Lang>
</Depth>

## [walk] Scan one rule

`cpu>=0.9 and env=="prod eu"`, token by token. Step through it with the numbered buttons.

<Depth level={3}>
Taking the longest match is called <Term id="maximal-munch">maximal munch</Term>. It explains a few classic surprises. In C, `x+++y` lexes as `x ++ + y`, never `x + ++y`. Older C++ needed a space in `vector<vector<int> >`, because `>>` was always one token, until C++11 changed the rule. Keeping a <Term id="source-span">source span</Term> on every token costs two integers, and every later error message depends on it.
</Depth>

## [again] On-call writes the rule again

The same rule, through the lexer. Seven tokens come out, the same seven as for the spaced-out version, and the parser accepts the rule. A rule that really is broken now says where, for example `1:8: unterminated string`.

<Depth level={2}>
Make the error token stop at the end of the line. One missing quote then costs one error, and the lexer still reports problems on the following lines in the same run.
</Depth>

<Depth level={3}>
Tokens are also the right place to keep what tools need later. A formatter wants comments, so it asks the lexer to keep them as trivia attached to the next token. Roslyn, the C# compiler, attaches comments and whitespace to tokens as trivia, which lets refactorings rewrite code without losing a comment.
</Depth>

## [play] Scan a rule yourself

Pick a rule or type one, then step the scanner and watch where each token ends. Compare its tokens with the words that `split()` produces, then break the scanner by turning off longest match.

### What to keep on every token

Kind, start and end offsets, and nothing else that can be recomputed. The text is a slice of the source. Line and column can be computed from the offset when an error is printed, which is what Go's `token.FileSet` does. If tools will rewrite the code, also keep the whitespace and comments around each token.
